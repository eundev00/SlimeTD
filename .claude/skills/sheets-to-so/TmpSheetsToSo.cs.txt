using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;

// sheets-to-so 스킬이 Assets/Scripts/Editor/TmpSheetsToSo.cs로 복사해 배치 모드로 실행하고 지우는 임시 스크립트
public static class TmpSheetsToSo
{
    private const string LogTag = "TmpSheetsToSo";
    private const string SlimeFolder = "Assets/Datas/Slimes";
    private const string StageFolder = "Assets/Datas/Stages";
    private const string GameConfigPath = "Assets/Datas/GameConfig.asset";
    private const string LobbyScenePath = "Assets/Scenes/LobbyScene.unity";

    private static int _errors;

    public static void Run()
    {
        _errors = 0;

        TowerSheetImporter.Import();
        SkillSheetImporter.Import();
        HunterCardGachaSheetImporter.Import();

        var slimes = ImportSlimes();
        ImportStages(slimes);
        ImportGameConfig();

        AssetDatabase.SaveAssets();
        ResourceKeysGenerator.CreateDataKeys();
        ConnectLobbyStage();
        TmpFontTrimmer.Trim();

        Debug.Log($"[{LogTag}] 완료, 오류 {_errors}개");
    }

    private static Dictionary<int, SlimeDataBase> ImportSlimes()
    {
        var result = new Dictionary<int, SlimeDataBase>();
        if (!SheetReader.TryReadTable(SheetReader.PathOf("Slimes.csv"), LogTag, out var rows))
        {
            _errors++;
            return result;
        }

        foreach (var row in rows)
        {
            string codeName = SheetReader.Get(row, "codeName");
            string kind = SheetReader.Get(row, "kind");
            if (!SheetReader.TryGetInt(row, "slimeId", out int slimeId) || string.IsNullOrEmpty(codeName))
            {
                Error($"슬라임 행을 읽지 못했습니다: {codeName}");
                continue;
            }

            bool boss = kind == nameof(SlimeKind.Boss);
            string path = $"{SlimeFolder}/{codeName}.asset";
            var asset = AssetDatabase.LoadAssetAtPath<SlimeDataBase>(path);
            if (asset != null && (asset is BossSlimeData) != boss)
            {
                Error($"{path}의 종류가 시트(kind={kind})와 다릅니다. 에셋을 지우고 다시 만들어야 합니다 (프리팹 연결은 다시 해야 함)");
                continue;
            }

            if (asset == null)
            {
                asset = boss ? ScriptableObject.CreateInstance<BossSlimeData>() : ScriptableObject.CreateInstance<SlimeData>();
                AssetDatabase.CreateAsset(asset, path);
                Debug.Log($"[{LogTag}] 슬라임 에셋을 만들었습니다. 프리팹은 인스펙터에서 연결하세요: {path}", asset);
            }

            var so = new SerializedObject(asset);
            so.FindProperty("_slimeId").intValue = slimeId;
            so.FindProperty("_baseHealth").intValue = ReadInt(row, "baseHealth");
            so.FindProperty("_baseSpeed").floatValue = ReadFloat(row, "baseSpeed");
            so.FindProperty("_lifeCost").intValue = ReadInt(row, "lifeCost");
            so.FindProperty("_jellyReward").intValue = ReadInt(row, "jellyReward");
            so.FindProperty("_renderGroup").intValue = (int)ReadEnum<SlimeRenderGroup>(row, "renderGroup");
            if (boss)
                so.FindProperty("_instantGameOver").boolValue = ReadInt(row, "instantGameOver") == 1;
            so.ApplyModifiedPropertiesWithoutUndo();
            EditorUtility.SetDirty(asset);

            if (asset.Prefab == null)
                Debug.Log($"[{LogTag}] 프리팹이 연결되지 않은 슬라임: {path}", asset);

            result[slimeId] = asset;
        }

        Debug.Log($"[{LogTag}] 슬라임 {result.Count}개");
        return result;
    }

    private static void ImportStages(Dictionary<int, SlimeDataBase> slimes)
    {
        if (!SheetReader.TryReadTable(SheetReader.PathOf("Stages.csv"), LogTag, out var rows))
        {
            _errors++;
            return;
        }

        if (!AssetDatabase.IsValidFolder(StageFolder))
            AssetDatabase.CreateFolder("Assets/Datas", "Stages");

        foreach (var row in rows)
        {
            string codeName = SheetReader.Get(row, "codeName");
            string path = $"{StageFolder}/Stage_{codeName}.asset";
            var asset = AssetDatabase.LoadAssetAtPath<StageData>(path);
            if (asset == null)
            {
                asset = ScriptableObject.CreateInstance<StageData>();
                AssetDatabase.CreateAsset(asset, path);
            }

            var so = new SerializedObject(asset);
            so.FindProperty("_stageId").intValue = ReadInt(row, "stageId");
            so.FindProperty("_codeName").stringValue = codeName;
            so.FindProperty("_mode").intValue = (int)ReadEnum<GameMode>(row, "mode");
            so.FindProperty("_difficulty").intValue = (int)ReadEnum<Difficulty>(row, "difficulty");
            so.FindProperty("_displayName").stringValue = SheetReader.Get(row, "name");
            so.FindProperty("_enabled").boolValue = ReadInt(row, "enabled") == 1;
            so.FindProperty("_startingLife").intValue = ReadInt(row, "startingLife");
            so.FindProperty("_startingJelly").intValue = ReadInt(row, "startingJelly");
            so.FindProperty("_spawnInterval").floatValue = ReadFloat(row, "spawnInterval");
            so.FindProperty("_interWaveDelay").floatValue = ReadFloat(row, "interWaveDelay");
            so.FindProperty("_normalHpGrowth").floatValue = ReadFloat(row, "normalHpGrowth");
            FillWaves(so.FindProperty("_waves"), codeName, slimes);
            so.ApplyModifiedPropertiesWithoutUndo();
            EditorUtility.SetDirty(asset);

            Debug.Log($"[{LogTag}] 스테이지 {codeName}: 최종 웨이브 {asset.FinalWave}");
        }
    }

    private static void FillWaves(SerializedProperty wavesProp, string codeName, Dictionary<int, SlimeDataBase> slimes)
    {
        wavesProp.arraySize = 0;
        if (!SheetReader.TryReadTable(SheetReader.PathOf($"Waves/{codeName}.csv"), LogTag, out var rows))
        {
            _errors++;
            return;
        }

        var waves = rows
            .GroupBy(r => ReadInt(r, "wave"))
            .OrderBy(g => g.Key)
            .ToList();

        wavesProp.arraySize = waves.Count;
        for (int w = 0; w < waves.Count; w++)
        {
            if (waves[w].Key != w + 1)
                Error($"{codeName} 웨이브 번호가 1부터 이어지지 않습니다: {waves[w].Key}");

            var groupsProp = wavesProp.GetArrayElementAtIndex(w).FindPropertyRelative("_groups");
            var groups = waves[w].OrderBy(r => ReadInt(r, "order")).ToList();
            groupsProp.arraySize = groups.Count;

            for (int g = 0; g < groups.Count; g++)
            {
                var groupProp = groupsProp.GetArrayElementAtIndex(g);
                int slimeId = ReadInt(groups[g], "slimeId");
                if (!slimes.TryGetValue(slimeId, out var slime))
                    Error($"{codeName} 웨이브 {waves[w].Key}: 슬라임 {slimeId}가 없습니다");

                groupProp.FindPropertyRelative("_slime").objectReferenceValue = slime;
                groupProp.FindPropertyRelative("_count").intValue = ReadInt(groups[g], "count");
                groupProp.FindPropertyRelative("_delay").floatValue = SheetReader.TryGetFloat(groups[g], "delay", out float delay) ? delay : 0f;
            }
        }
    }

    private static void ImportGameConfig()
    {
        var config = AssetDatabase.LoadAssetAtPath<GameConfig>(GameConfigPath);
        if (config == null)
        {
            Error($"GameConfig가 없습니다: {GameConfigPath}");
            return;
        }

        if (!SheetReader.TryReadTable(SheetReader.PathOf("GameConstants.csv"), LogTag, out var constantRows)
            || !SheetReader.TryReadTable(SheetReader.PathOf("Items.csv"), LogTag, out var itemRows)
            || !SheetReader.TryReadTable(SheetReader.PathOf("NewAccount.csv"), LogTag, out var accountRows))
        {
            _errors++;
            return;
        }

        var constants = constantRows.ToDictionary(r => SheetReader.Get(r, "key"), r => SheetReader.Get(r, "value"));
        var so = new SerializedObject(config);
        so.FindProperty("_summonBaseCost").intValue = ParseInt(constants, "summonBaseCost");
        so.FindProperty("_summonCostIncrease").intValue = ParseInt(constants, "summonCostIncrease");
        so.FindProperty("_summonGradeId").intValue = ParseInt(constants, "summonGradeId");
        so.FindProperty("_slowMinSpeedRatio").floatValue = ParseFloat(constants, "slowMinSpeedRatio");
        so.FindProperty("_bossSlowScale").floatValue = ParseFloat(constants, "bossSlowScale");

        // 새 계정 지급은 서버가 할 일이지만 서버 전까지는 로컬 저장소가 GameConfig 값으로 계정을 만든다
        var codeNames = itemRows.ToDictionary(r => ReadInt(r, "itemId"), r => SheetReader.Get(r, "codeName"));
        int gold = 0, gem = 0;
        foreach (var row in accountRows)
        {
            codeNames.TryGetValue(ReadInt(row, "itemId"), out string codeName);
            if (codeName == "Gold") gold = ReadInt(row, "amount");
            else if (codeName == "Gem") gem = ReadInt(row, "amount");
            else Debug.Log($"[{LogTag}] 클라이언트가 아직 지급하지 못하는 새 계정 아이템: {codeName}");
        }

        so.FindProperty("_initialLobbyGold").intValue = gold;
        so.FindProperty("_initialGem").intValue = gem;
        so.ApplyModifiedPropertiesWithoutUndo();
        EditorUtility.SetDirty(config);
    }

    private static void ConnectLobbyStage()
    {
        // 씬을 Single로 열면 사용하지 않는 에셋이 언로드되므로 씬을 먼저 열고 에셋을 읽는다
        var scene = EditorSceneManager.OpenScene(LobbyScenePath, OpenSceneMode.Single);
        var stage = AssetDatabase.LoadAssetAtPath<StageData>(DataKeys.GetDataPath(DataKeys.Stage_SingleNormal));
        var scope = UnityEngine.Object.FindAnyObjectByType<LobbyLifetimeScope>();
        if (scope == null)
        {
            Error("LobbyScene에서 LobbyLifetimeScope를 찾지 못했습니다");
            return;
        }

        var so = new SerializedObject(scope);
        var prop = so.FindProperty("_stage");
        if (prop.objectReferenceValue == stage)
            return;

        prop.objectReferenceValue = stage;
        so.ApplyModifiedPropertiesWithoutUndo();
        EditorSceneManager.MarkSceneDirty(scene);
        EditorSceneManager.SaveScene(scene);
        Debug.Log($"[{LogTag}] 로비 스테이지 연결: {stage.name}");
    }

    private static int ReadInt(Dictionary<string, string> row, string key)
    {
        if (SheetReader.TryGetInt(row, key, out int value))
            return value;

        if (!string.IsNullOrEmpty(SheetReader.Get(row, key)))
            Error($"정수가 아닙니다: {key}='{SheetReader.Get(row, key)}'");
        return 0;
    }

    private static float ReadFloat(Dictionary<string, string> row, string key)
    {
        if (SheetReader.TryGetFloat(row, key, out float value))
            return value;

        Error($"숫자가 아닙니다: {key}='{SheetReader.Get(row, key)}'");
        return 0f;
    }

    private static T ReadEnum<T>(Dictionary<string, string> row, string key) where T : struct
    {
        string text = SheetReader.Get(row, key);
        if (Enum.TryParse(text, false, out T value) && Enum.IsDefined(typeof(T), value))
            return value;

        Error($"{typeof(T).Name}에 없는 값입니다: {key}='{text}'");
        return default;
    }

    private static int ParseInt(Dictionary<string, string> map, string key)
    {
        if (map.TryGetValue(key, out string text) && int.TryParse(text, out int value))
            return value;

        Error($"GameConstants 키가 없거나 정수가 아닙니다: {key}");
        return 0;
    }

    private static float ParseFloat(Dictionary<string, string> map, string key)
    {
        if (map.TryGetValue(key, out string text) && float.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out float value))
            return value;

        Error($"GameConstants 키가 없거나 숫자가 아닙니다: {key}");
        return 0f;
    }

    private static void Error(string message)
    {
        _errors++;
        Debug.Log($"[{LogTag}] 오류: {message}");
    }
}
